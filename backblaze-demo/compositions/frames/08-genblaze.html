              providers="gmi-cloud+elevenlabs")
    .step(GMICloudImageProvider(),  # thumbnail
          model="seedream-5.0-lite", modality="IMAGE")
    .step(ElevenLabsTTSProvider(),  # soundscape
          model="sound-fx", duration=10)
    .step(ElevenLabsTTSProvider(),  # voice
          model="eleven_v3", voice_id=voice)
    .run(sink=storage, timeout=180)
)
